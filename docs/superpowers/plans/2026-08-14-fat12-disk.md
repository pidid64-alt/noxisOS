# FAT12 Runtime Disk Read/Write Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let the noxis 32-bit kernel read and write FAT12 files on a floppy at runtime from the shell, via a BIOS real-mode thunk.

**Architecture:** A 16-bit assembly thunk (`realmode_disk.asm`) performs a protected-mode→real-mode→protected-mode switch and calls BIOS `int 13h` to read/write raw sectors, copying data through a reserved low-memory scratch buffer at `0x9000`–`0xB000`. A FAT12 layer (`fs_fat12.c`) sits on top, and the shell gains `LS`, `CAT`, `WRITE` commands. The data volume is a **second floppy** (`-fdb floppy.img`) built reproducibly by the Makefile.

**Tech Stack:** i686 32-bit freestanding C (`-ffreestanding -m32 -Wall -Wextra -fno-pie -fno-stack-protector -fcommon`), NASM 16/32-bit assembly, GNU ld (`elf_i386`), qemu-system-i386, `mkfs.fat`/`mcopy` (or loop-mount fallback) for image creation.

## Global Constraints

- All C compiles with `CFLAGS = -g -ffreestanding -Wall -Wextra -fno-exceptions -fno-pie -fno-stack-protector -fcommon -m32` (verbatim from `23-fixes/Makefile`).
- Kernel is linked at `-Ttext 0x1000`; boot sector loads it at `0x1000` (31 sectors ≈ up to `0x4E00`).
- **Scratch buffer region `0x9000`–`0xB000` is free after boot** — the boot sector uses `0x9000` only as its real-mode stack *before* the PM switch; `switch_pm.asm` then moves the PM stack to `0x90000`. The kernel never touches `0x9000`. No bootloader change is required to "reserve" it. Do NOT use `0x9000` as a PM stack.
- VGA text buffer is `0xB8000` — keep the scratch region below `0xB000`.
- FAT12 geometry: 512 B/sector, 1.44 MB, 2 heads, 18 sec/track, 224 root entries (14 sectors), 2 FAT copies.
- Filenames are 8.3, uppercase (`hello.txt` → `HELLO.TXT`).
- Existing commands (`HELP CLEAR ECHO VERSION TIME END PAGE UPTIME`) MUST remain unchanged.
- Each task ends with a build + boot verification and a commit.

## Test approach (no unit framework)

This is a bare-metal OS; "tests" are: (1) for pure helper functions, a host-compiled `tests/` C program; (2) for kernel behavior, `make` then boot qemu with both floppies and drive the shell via the qemu monitor, observing VGA output. The qemu driver snippet in Task 8 is reused for every boot test.

```nasm
; 16-bit code segment for the real-mode thunk. base=0, limit=0xffff,
; present, ring0, code, 16-bit default operand size (D bit = 0).
gdt_rmcode:
    dw 0xffff
    dw 0x0
    db 0x0
    db 10011010b ; 0x9A: present, ring0, code, readable
    db 00001111b ; limit bits 16-19 = 0xF; D bit (bit 6 of flags) = 0 => 16-bit
    db 0x0

gdt_rmdata:
    dw 0xffff
    dw 0x0
    db 0x0
    db 10010010b ; 0x92: present, ring0, data, writable
    db 00001111b
    db 0x0

gdt_end:
```
Then update the two `equ` lines at the bottom to also export the new offsets:
```nasm
CODE_SEG   equ gdt_code   - gdt_start
DATA_SEG   equ gdt_data   - gdt_start
RM_CODE_SEG equ gdt_rmcode  - gdt_start
RM_DATA_SEG equ gdt_rmdata  - gdt_start
```
Note: `gdt_end` label is already after `gdt_data` in the original; in this edit `gdt_end` moves to after `gdt_rmdata`. Keep the descriptor `dw gdt_end - gdt_start - 1`.

  Both copy `count*512` bytes between `pm_buf` (anywhere in PM) and the disk via the scratch buffer at `0x9000`. `lba` is 0-based logical sector (LBA). Drive is the data floppy = `0x81` (second floppy).

- [ ] **Step 2: Write `realmode_disk.asm`**

The thunk: from PM, save context, switch to 16-bit RM using `RM_CODE_SEG`/`RM_DATA_SEG`, set up a small RM stack, copy the caller's PM buffer into the scratch area at `0x9000`, build the `int 13h` register frame, call `int 0x13`, copy scratch back to the PM buffer (for reads), reload PM segments, far-return. LBA→CHS conversion for 1.44 MB (18 sec/track, 2 heads):
```
cylinder = lba / 36
temp     = lba % 36
head     = temp / 18
sector   = (temp % 18) + 1
```
The C entry points are `disk_read` / `disk_write` (cdecl, 3 args: lba=stack[4], count=stack[8], pm_buf=stack[12]).

```nasm
[bits 32]
global disk_read
global disk_write

SCRATCH   equ 0x9000      ; 15 KB scratch buffer (0x9000 .. 0xB000-ish)
RM_STACK  equ 0x8F00      ; tiny real-mode stack just below scratch

; Convert LBA (in EAX) to CHS. Output: CH=cyl, CL=sector(6..0)|cyl(7..6), DH=head
; uses EBX as scratch.
%macro lba_to_chs 0
    xor edx, edx
    mov ebx, 36
    div ebx              ; eax=cyl, edx=temp
    mov ch, al           ; cylinder low 8 bits
    mov bl, 18
    mov eax, edx
    xor edx, edx
    div ebx              ; eax=head, edx=sector-1
    mov dh, al           ; head
    mov cl, dl
    inc cl               ; sector (1-based)
    shl ch, 6            ; cylinder high 2 bits into cl[7:6]
    or  cl, ch
    mov ch, al_cyl       ; placeholder replaced below
%endmacro
```
> NOTE: the macro above is illustrative; implement inline (see below) to avoid label collisions.

Full implementation (replace the macro with inline code):
```nasm
[bits 32]
global disk_read
global disk_write

SCRATCH  equ 0x9000
RM_STACK equ 0x8F00

; Far pointer to 16-bit thunk, filled at runtime from RM_CODE_SEG.
; We compute it as (RM_CODE_SEG << 16) | rm_entry.

disk_read:
    push ebp
    mov  ebp, esp
    pushad
    mov  edi, [ebp+16]   ; pm_buf
    mov  ecx, [ebp+12]   ; count
    mov  eax, [ebp+8]    ; lba
    mov  byte [rm_op], 0x02   ; 0x02 = read
    jmp  do_rm

disk_write:
    push ebp
    mov  ebp, esp
    pushad
    mov  edi, [ebp+16]   ; pm_buf
    mov  ecx, [ebp+12]   ; count
    mov  eax, [ebp+8]    ; lba
    mov  byte [rm_op], 0x03   ; 0x03 = write
    ; copy PM buffer -> scratch first (for writes)
    push ecx
    shl  ecx, 9          ; count*512
    mov  esi, edi
    mov  edi, SCRATCH
    cld
    rep movsb
    pop  ecx

do_rm:
    ; EAX=lba, ECX=count, EDI=pm_buf (for read copy-back)
    ; set DS/ES to kernel data for accessing rm_op etc.
    mov  ax, DATA_SEG
    mov  ds, ax
    mov  es, ax

    ; compute CHS
    mov  ebx, 36
    xor  edx, edx
    div  ebx            ; eax=cyl, edx=temp
    mov  [cyl], al
    mov  eax, edx
    xor  edx, edx
    mov  ebx, 18
    div  ebx            ; eax=head, edx=sector-1
    mov  [head], al
    mov  [sect], dl
    inc  byte [sect]
    mov  al, [cyl]
    shl  al, 6
    or   [sect], al

    ; copy SCRATCH-> (for write already done) ; for read, scratch is the dst
    ; build the 16-bit far call into RM segment
    mov  eax, RM_CODE_SEG
    shl  eax, 16
    mov  ax, rm16
    mov  [rm_far], eax

    ; switch to real mode: reload segments to RM_DATA_SEG, set RM stack
    mov  ax, RM_DATA_SEG
    mov  ds, ax
    mov  es, ax
    mov  ss, ax
    mov  esp, RM_STACK

    ; clear PE bit in CR0
    mov  eax, cr0
    and  eax, 0xFFFFFFFE
    mov  cr0, eax

    ; far jump to 16-bit code segment to flush prefetch
    jmp  far [rm_far]

[bits 16]
rm16:
    ; we are in real mode now, 16-bit
    mov  ax, RM_DATA_SEG
    mov  ds, ax
    mov  es, ax
    mov  ss, ax
    mov  sp, RM_STACK

    ; set up int 13h registers
    mov  ah, [cs:rm_op]   ; but we are not in a code segment with cs=RM; use ds
    ; rm_op is in the PM-loaded data; reload via known segment
    mov  dl, 0x81         ; second floppy
    mov  al, [count8]     ; sectors
    mov  ch, [cyl8]       ; cylinder low
    mov  cl, [sect8]      ; sector | cyl high
    mov  dh, [head8]      ; head
    mov  bx, SCRATCH & 0xFFFF  ; ES:BX = scratch
    mov  es, bx           ; NOTE: set es to segment of scratch
    mov  bx, 0

    int  0x13
    mov  [rm_status], ah  ; save BIOS status

    ; switch back to protected mode
    mov  eax, cr0
    or   eax, 0x1
    mov  cr0, eax
    jmp  CODE_SEG:back_pm

[bits 32]
back_pm:
    mov  ax, DATA_SEG
    mov  ds, ax
    mov  es, ax
    mov  ss, ax
    mov  esp, ebp        ; restore PM stack
    ; for read: copy SCRATCH -> pm_buf (EDI still holds pm_buf)
    cmp  byte [rm_op], 0x02
    jne  done_copy
    mov  ecx, [count512]
    mov  esi, SCRATCH
    mov  edi, [pm_buf_save]
    cld
    rep movsb
done_copy:
    mov  al, [rm_status]
    mov  [rm_status_pm], al
    popad
    mov  eax, 0          ; DISK_OK; (caller checks rm_status_pm separately)
    ; return DISK_OK/ERR based on rm_status
    cmp  byte [rm_status], 0
    jne  .err
    pop  ebp
    ret
.err:
    mov  eax, 1          ; DISK_ERR
    pop  ebp
    ret

; storage for values passed between 32/16-bit halves
rm_op        db 0
count8       db 0
cyl8         db 0
sect8        db 0
head8        db 0
count512     dd 0
pm_buf_save  dd 0
rm_status    db 0
rm_status_pm db 0
rm_far       dd 0
```
> Implementation notes for the implementer:
> - `rm_op`, `count8`, `cyl8`, `sect8`, `head8`, `count512`, `pm_buf_save`, `rm_status`, `rm_status_pm`, `rm_far` are 32-bit-data-section variables (place them in `.data`/bss as appropriate for the freestanding link; since the kernel is a flat binary at 0x1000, plain labels in the same object work).
> - `count512` must be set = `count*512` before `do_rm` (add `mov eax, ecx; shl eax,9; mov [count512], eax` and `mov [pm_buf_save], edi`).
> - `es` for `int 0x13` must point to the segment containing `SCRATCH` (0x9000 → segment `0x900`). Set `mov ax, 0x900; mov es, ax` then `mov bx, 0`.
> - The `jmp far [rm_far]` requires `rm_far` = `RM_CODE_SEG:rm16` packed as offset:seg = `rm16`(low) : `RM_CODE_SEG`(high). Build it as `mov word [rm_far], rm16; mov word [rm_far+2], RM_CODE_SEG`.
> - `lgdt` does NOT need re-issuing: the GDT already in memory now includes `RM_CODE_SEG`/`RM_DATA_SEG`. The thunk only changes segment registers and CR0.PE.
> - Reading `rm_op` in 16-bit code: it lives in the same flat 32-bit data region; with `ds=RM_DATA_SEG` (base 0) the offset equals the linear address, so direct `[rm_op]` works.

- [ ] **Step 1: Write `fs_to_83` and `fs_init`**

```c
#include "fs_fat12.h"
#include "../cpu/realmode_disk.h"
#include "../drivers/screen.h"
#include "../libc/string.h"
#include "../libc/mem.h"
#include <stdint.h>

/* cached geometry, filled by fs_init() */
static uint32_t fat_lba;      /* LBA of first FAT */
static uint32_t fat_size;     /* sectors per FAT */
static uint32_t root_lba;     /* LBA of root dir */
static uint32_t root_entries; /* number of 32-byte root entries */
static uint32_t data_start;   /* LBA of cluster 2 */

static uint8_t boot_sector[SECTOR_SIZE];

uint8_t fs_to_83(const char *name, char out83[11]) {
    int i, p = 0;
    char *dot = 0;
    for (i = 0; name[i]; i++) if (name[i] == '.') dot = (char*)&name[i];
    /* name part */
    int name_len = dot ? (int)(dot - name) : (int)strlen(name);
    if (name_len == 0 || name_len > 8) return 0;
    for (i = 0; i < 8; i++) {
        if (i < name_len) out83[p++] = (name[i] >= 'a' && name[i] <= 'z')
                                          ? name[i] - 32 : name[i];
        else out83[p++] = ' ';
    }
    /* ext part */
    if (dot) {
        const char *e = dot + 1;
        int el = (int)strlen(e);
        if (el > 3) return 0;
        for (i = 0; i < 3; i++) {
            if (i < el) out83[p++] = (e[i] >= 'a' && e[i] <= 'z')
                                         ? e[i] - 32 : e[i];
            else out83[p++] = ' ';
        }
    } else {
        for (i = 0; i < 3; i++) out83[p++] = ' ';
    }
    return 1;
}

void fs_init(void) {
    if (disk_read(0, 1, boot_sector) != DISK_OK) {
        kprint("FS: cannot read boot sector of data volume\n");
        return;
    }
    /* BPB is at offset 11 (bytesPerSector), 13 (sectorsPerCluster),
       14 (reservedSectors), 16 (fats), 17 (rootEntries),
       36 (fatSectors32) / 22 (fatSectors16) */
    uint16_t rsv  = *(uint16_t*)&boot_sector[14];
    uint8_t  fats = boot_sector[16];
    uint16_t spf  = *(uint16_t*)&boot_sector[22];
    root_entries  = *(uint16_t*)&boot_sector[17];
    fat_lba     = rsv;
    fat_size    = spf;
    root_lba    = rsv + (uint32_t)fats * spf;
    uint32_t root_secs = (root_entries * 32 + SECTOR_SIZE - 1) / SECTOR_SIZE;
    data_start  = root_lba + root_secs;
}
```

- [ ] **Step 2: Write `fs_open`, `fs_read`, `fs_list_root`**

```c
static uint32_t cluster_to_lba(uint32_t c) { return data_start + (c - 2); }

static uint16_t fat_entry(uint32_t cluster) {
    /* 12-bit entry at offset cluster*3/2 in the FAT */
    uint32_t off = cluster * 3 / 2;
    static uint8_t buf[SECTOR_SIZE];
    uint32_t sector = fat_lba + off / SECTOR_SIZE;
    disk_read(sector, 1, buf);
    uint16_t lo = buf[off % SECTOR_SIZE];
    uint16_t hi = buf[(off % SECTOR_SIZE) + 1];
    uint16_t val = (hi << 8) | lo;
    return (cluster & 1) ? (val >> 4) : (val & 0x0FFF);
}

int fs_open(const char *name83, uint32_t *out_first_cluster, uint32_t *out_size) {
    uint8_t dir[SECTOR_SIZE];
    uint32_t root_secs = (root_entries * 32 + SECTOR_SIZE - 1) / SECTOR_SIZE;
    for (uint32_t s = 0; s < root_secs; s++) {
        if (disk_read(root_lba + s, 1, dir) != DISK_OK) return FS_ERR;
        for (uint32_t e = 0; e < SECTOR_SIZE; e += 32) {
            uint8_t *ent = &dir[e];
            if (ent[0] == 0x00) return FS_NOT_FOUND; /* end of dir */
            if (ent[0] == 0xE5) continue;            /* deleted */
            if (ent[11] & 0x0F) continue;            /* not a file (volume/etc) */
            if (memcmp(ent, name83, 11) == 0) {
                *out_first_cluster = *(uint16_t*)&ent[26];
                *out_size = *(uint32_t*)&ent[28];
                return FS_OK;
            }
        }
    }
    return FS_NOT_FOUND;
}

int fs_read(uint32_t first_cluster, uint32_t size, char *out_buf, uint32_t buf_len) {
    if (size > buf_len) return FS_ERR;
    if (size > SCRATCH_BYTES) return FS_ERR;
    uint32_t remaining = size;
    uint32_t cluster = first_cluster;
    static uint8_t cluster_buf[SECTOR_SIZE];
    uint32_t spc = 1; /* sectors per cluster from BPB[13]; assume 1 for 1.44MB default,
                         read boot_sector[13] in fs_init for correctness */
    while (remaining > 0 && cluster < 0xFF8) {
        uint32_t lba = cluster_to_lba(cluster);
        uint32_t to_read = remaining > SECTOR_SIZE ? SECTOR_SIZE : remaining;
        if (disk_read(lba, 1, cluster_buf) != DISK_OK) return FS_ERR;
        memcpy(out_buf, cluster_buf, to_read);
        out_buf += to_read;
        remaining -= to_read;
        if (remaining == 0) break;
        cluster = fat_entry(cluster);
    }
    return FS_OK;
}

int fs_list_root(void (*cb)(const char *name83, uint32_t size)) {
    uint8_t dir[SECTOR_SIZE];
    uint32_t root_secs = (root_entries * 32 + SECTOR_SIZE - 1) / SECTOR_SIZE;
    for (uint32_t s = 0; s < root_secs; s++) {
        if (disk_read(root_lba + s, 1, dir) != DISK_OK) return FS_ERR;
        for (uint32_t e = 0; e < SECTOR_SIZE; e += 32) {
            uint8_t *ent = &dir[e];
            if (ent[0] == 0x00) return FS_OK;
            if (ent[0] == 0xE5) continue;
            if (ent[11] & 0x0F) continue;
            char nm[11];
            memcpy(nm, ent, 11);
            uint32_t sz = *(uint32_t*)&ent[28];
            cb(nm, sz);
        }
    }
    return FS_OK;
}
```

- [ ] **Step 3: Write `fs_write` (allocate clusters, write FAT + dir)**

```c
int fs_write(const char *name83, const char *data, uint32_t len) {
    if (len > SCRATCH_BYTES) return FS_NO_SPACE;
    uint32_t nclusters = (len + SECTOR_SIZE - 1) / SECTOR_SIZE;
    if (nclusters == 0) nclusters = 1;

    /* scan FAT for nclusters free (0x000) entries, building a chain */
    static uint8_t fatbuf[SECTOR_SIZE];
    uint32_t chain[32]; /* up to 32 clusters (<=16KB) */
    uint32_t found = 0;
    uint32_t fat_secs = fat_size; /* span of one FAT copy */
    for (uint32_t c = 2; c < 0xFF8 && found < nclusters; c++) {
        uint16_t fe = fat_entry(c);
        if (fe == 0x000) {
            chain[found++] = c;
        }
    }
    if (found < nclusters) return FS_NO_SPACE;

    /* link the chain */
    for (uint32_t i = 0; i < found; i++) {
        uint16_t val = (i + 1 < found) ? (uint16_t)chain[i+1] : 0xFFF;
        /* write 12-bit entry into fatbuf then flush */
        uint32_t off = chain[i] * 3 / 2;
        uint32_t sector = fat_lba + off / SECTOR_SIZE;
        disk_read(sector, 1, fatbuf);
        uint16_t lo = fatbuf[off % SECTOR_SIZE];
        uint16_t hi = fatbuf[(off % SECTOR_SIZE)+1];
        uint16_t cur = (hi << 8) | lo;
        if (chain[i] & 1) cur = (cur & 0x000F) | (val << 4);
        else             cur = (cur & 0xF000) | (val & 0x0FFF);
        fatbuf[off % SECTOR_SIZE]   = cur & 0xFF;
        fatbuf[(off % SECTOR_SIZE)+1] = (cur >> 8) & 0xFF;
        disk_write(sector, 1, fatbuf);
    }
    /* write data clusters */
    for (uint32_t i = 0; i < found; i++) {
        uint32_t off = i * SECTOR_SIZE;
        uint8_t blk[SECTOR_SIZE];
        for (int b = 0; b < SECTOR_SIZE; b++)
            blk[b] = (off + b < len) ? (uint8_t)data[off + b] : 0;
        disk_write(cluster_to_lba(chain[i]), 1, blk);
    }
    /* write/update root dir entry */
    uint8_t dir[SECTOR_SIZE];
    uint32_t root_secs = (root_entries * 32 + SECTOR_SIZE - 1) / SECTOR_SIZE;
    uint32_t slot_sector = 0, slot_off = 0; int found_slot = 0;
    for (uint32_t s = 0; s < root_secs && !found_slot; s++) {
        disk_read(root_lba + s, 1, dir);
        for (uint32_t e = 0; e < SECTOR_SIZE; e += 32) {
            if (dir[e] == 0x00 || dir[e] == 0xE5) { slot_sector = s; slot_off = e; found_slot = 1; break; }
            if (memcmp(&dir[e], name83, 11) == 0) { slot_sector = s; slot_off = e; found_slot = 1; break; }
        }
    }
    if (!found_slot) return FS_NO_SPACE;
    disk_read(root_lba + slot_sector, 1, dir);
    memcpy(&dir[slot_off], name83, 11);
    dir[slot_off+11] = 0x20; /* archive */
    *(uint16_t*)&dir[slot_off+26] = (uint16_t)chain[0];
    *(uint32_t*)&dir[slot_off+28] = len;
    disk_write(root_lba + slot_sector, 1, dir);
    return FS_OK;
}
```

> NOTE: `sectors per cluster` (BPB[13]) for a 1.44 MB floppy is 1, so `spc=1` is correct; read it in `fs_init` (`boot_sector[13]`) and use it in `fs_read`/`cluster_to_lba` if you want full generality. For the standard image we build (Task 7), spc=1.

```c
    kprint("Volume contents:\n");

        int i = 0;
        for (int k = 0; k < 11; k++) line[i++] = name83[k];

int disk_cat(const char *name, char *buf, uint32_t buflen) {
    char n83[11];
    if (!fs_to_83(name, n83)) { kprint("Bad filename (need 8.3)\n"); return -1; }
```

```c
static void cmd_ls(char *args) {
    UNUSED(args);
    disk_ls();
}

static void cmd_cat(char *args) {
    if (!args || args[0] == '\0') { kprint("Usage: CAT <name>\n"); return; }
    static char buf[4096];
    int n = disk_cat(args, buf, sizeof(buf));
    if (n > 0) {
        buf[n] = '\0';
        kprint(buf);
        if (buf[n-1] != '\n') kprint("\n");
    }
}

static void cmd_write(char *args) {
    if (!args || args[0] == '\0') { kprint("Usage: WRITE <name> <text>\n"); return; }
    /* split name and rest-of-line */
    char *p = args;
    while (*p && *p != ' ') p++;
    if (*p == '\0') { kprint("Usage: WRITE <name> <text>\n"); return; }
    *p = '\0';
    char *text = p + 1;
    int r = disk_write_file(args, text, (uint32_t)strlen(text));
    if (r == FS_OK)        kprint("Wrote file\n");
    else if (r == FS_NO_SPACE) kprint("Disk full / file too large\n");
    else                   kprint("Write error\n");
}
```
